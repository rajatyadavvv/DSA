class Solution:
    def searchRange(self, nums: list[int], target: int) -> list[int]:
        def first(nums,target):
            s = 0
            e = len(nums) - 1
            ans = -1

            while s<=e:
                mid = s + (e-s)//2

                if nums[mid] == target:
                    ans = mid
                    e = mid - 1
                elif nums[mid] > target:
                    e = mid -1
                else:
                    s = mid + 1
            
            return ans 
        
        def last(nums,target):
            s = 0
            e = len(nums) - 1

            ans = -1

            while s <= e:
                mid = s + (e-s)//2

                if nums[mid] == target:
                    ans = mid
                    s = mid + 1
                elif nums[mid] > target:
                    e = mid - 1
                else:
                    s = mid + 1
            return ans
        return [first(nums,target),last(nums,target)]
