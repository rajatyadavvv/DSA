class Solution:
    def maxArea(self, height: list[int]) -> int:
        n = len(height)
        i = 0
        j = n - 1
        max_area = 0

        while i < j:
            curr_area = (min(height[i],height[j]) * (j - i))
            max_area = max(curr_area, max_area)

            if height[i] > height[j]:
                j -= 1
            else:
                i += 1
        
        return max_area
