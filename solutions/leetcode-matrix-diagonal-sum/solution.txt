class Solution:
    def diagonalSum(self, mat: list[list[int]]) -> int:
        n = len(mat)
        total = 0

        for i in range(n):
            total += mat[i][i]
        
        for i in range(n):
            if i == n - 1 - i:
                continue
            else:
                total += mat[i][n-1-i]
        return total
