class Solution:
    def maxProfit(self, prices: list[int]) -> int:
        buy = prices[0]
        max_p = 0
        for i in range(1,len(prices)):
            if prices[i] < buy:
                buy = prices[i]
            else :
                max_p = max(max_p, prices[i] - buy)
        return max_p
