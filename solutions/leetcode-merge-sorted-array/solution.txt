class Solution:
    def merge(self, nums1: list[int], m: int, nums2: list[int], n: int) -> None:
        n1 = nums1[:m]
        nums1.clear()
        i = 0
        j = 0

        while i < m and j < n:
            if n1[i] < nums2[j]:
                nums1.append(n1[i])
                i += 1
            else:
                nums1.append(nums2[j])
                j += 1
            
        while i < m:
            nums1.append(n1[i])
            i += 1
        while j < n:
            nums1.append(nums2[j])
            j += 1
