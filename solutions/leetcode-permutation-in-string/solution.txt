from collections import Counter
class Solution:
    def checkInclusion(self, s1: str, s2: str) -> bool:
        l1, l2 = len(s1), len(s2)

        s1_count = Counter(s1)
        window = Counter(s2[:l1])

        if s1_count == window:
            return True
        
        i = l1

        while i < l2:

            window[s2[i]] += 1
            window[s2[i-l1]] -= 1

            if window[s2[i-l1]] == 0:
                del window[s2[i-l1]]
            
            i += 1

            if s1_count == window:
                return True
            
        return False
