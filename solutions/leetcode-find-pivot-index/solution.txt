class Solution:
    def pivotIndex(self, nums: list[int]) -> int:
        n = len(nums)
        left_sum = 0
        right_sum = 0
        total_sum = 0

        for i in range(0,n):
            total_sum += nums[i]
        
        for i in range(0,n):
            right_sum = total_sum - left_sum - nums[i]
            if right_sum == left_sum:
                return i
            else:
                left_sum += nums[i]
        return -1
