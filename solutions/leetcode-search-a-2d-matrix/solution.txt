class Solution:
    def searchMatrix(self, matrix: list[list[int]], target: int) -> bool:
        m = len(matrix)
        n = len(matrix[0])

        start = 0
        end = m - 1

        while start <= end:
            mid = start + (end - start)//2

            if matrix[mid][0] <= target <=  matrix[mid][n-1]:
                s = 0
                e = n-1

                while s<=e:
                    m = s + (e-s)//2
                    if matrix[mid][m] == target:
                        return True
                    elif target < matrix[mid][m]:
                        e = m - 1
                    else:
                        s = m + 1
                return False
            elif target < matrix[mid][0]:
                end = mid - 1 
            else:
                start = mid + 1
        
        return False
