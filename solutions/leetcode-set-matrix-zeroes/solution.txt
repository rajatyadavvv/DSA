class Solution:
    def setZeroes(self, matrix: list[list[int]]) -> None:
        m = len(matrix)
        n = len(matrix[0])

        col = [0] * n
        row = [0] * m

        for i in range(m):
            for j in range(n):
                if matrix[i][j] == 0:
                    col[j] = 1
                    row[i] = 1
                
        for i in range(m):
            for j in range(n):
                if row[i] == 1 or col[j] == 1:
                    matrix[i][j] = 0
