class Solution:
    def transpose(self, matrix: list[list[int]]) -> list[list[int]]:
        row = len(matrix)
        col = len(matrix[0])

        result = [
            [0 for i in range(row)] for i in range(col)]
        

        for i in range(row):
            for j in range(col):
                result[j][i] = matrix[i][j] 

        return result
